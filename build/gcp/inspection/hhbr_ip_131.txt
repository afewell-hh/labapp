#!/bin/bash
LOG_FILE="/var/log/hedgehog-lab/hhbr-ip-setup.log"
mkdir -p "$(dirname "$LOG_FILE")"

log() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] $*" | tee -a "$LOG_FILE"; }

log "hhbr IP setup starting..."

# Wait for hhbr bridge (up to 10 minutes - VLAB takes ~3-5 min to start)
MAX_WAIT=600
ELAPSED=0
while [ $ELAPSED -lt $MAX_WAIT ]; do
    if ip link show hhbr &>/dev/null; then
        log "hhbr bridge detected after ${ELAPSED}s"
        break
    fi
    sleep 10
    ELAPSED=$((ELAPSED + 10))
    log "Waiting for hhbr bridge... ($ELAPSED/$MAX_WAIT seconds)"
done

if ! ip link show hhbr &>/dev/null; then
    log "ERROR: hhbr bridge not found after $MAX_WAIT seconds"
    exit 1
fi

# Add IP if not present
if ! ip addr show hhbr | grep -q "172.30.0.2"; then
    ip addr add 172.30.0.2/21 dev hhbr 2>/dev/null && log "Added 172.30.0.2/21 to hhbr" || log "Failed to add IP (may exist)"
else
    log "IP already configured"
fi

log "hhbr IP setup complete"
exit 0
